# 三层检测 + 智能判断
check_gateway_health() {
    local http_ok=0
    local process_ok=0
    local systemd_ok=0
    
    # L1: HTTP 检测 (最可靠)
    if check_http; then
        http_ok=1
        result+="http:OK "
    else
        result+="http:FAIL "
    fi
    
    # L2: 进程检测
    if check_process; then
        process_ok=1
        result+="process:OK "
    fi
    
    # L3: Systemd 检测 (最终判断)
    if check_systemd; then
        systemd_ok=1
        result+="systemd:OK "
    fi
    
    echo "$result"
    
    # 核心判断逻辑
    if [ $systemd_ok -eq 1 ]; then
        return 0    # systemd正常=Gateway正常
    fi
    
    # 必须 HTTP + 进程都失败才算失败
    if [ $http_ok -eq 0 ] && [ $process_ok -eq 0 ]; then
        return 1
    fi
    
    return 0
}
